async function loadCoaches() { try { // Fetch coaches ordered strictly by display_order let { data: coaches, error } = await supabaseClient .from('coaches') .select('*') .order('display_order', { ascending: true }); loadedCoaches = coaches || []; const container = document.getElementById('coachListContainer'); container.innerHTML = ''; if (loadedCoaches.length === 0) { container.innerHTML = `
No coaches registered yet.
`; return; } loadedCoaches.forEach((c, index) => { let photoDisplay = c.photo_url ? `` : `
🧢 No Coach Photo
`; container.innerHTML += `
${photoDisplay}

${c.name}

Position #${c.display_order || (index + 1)}
${c.role}

${c.bio || 'No bio provided.'}

`; }); } catch (err) { console.error("Coach load error:", err); } } async function changeCoachOrder(coachId, newPosition) { if (newPosition < 1) return; const { error } = await supabaseClient.from('coaches').update({ display_order: newPosition }).eq('id', coachId); if (error) alert('Error changing order: ' + error.message); else loadCoaches(); }
